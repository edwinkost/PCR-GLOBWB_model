#!/bin/bash

#SBATCH --job-name=et_validation
#SBATCH --qos=nf
#SBATCH --account=c3s24111     # adjust
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=4
#SBATCH --mem-per-cpu=24G
#SBATCH --time=06:00:00
#SBATCH --output=%x.%j.out
#SBATCH --error=%x.%j.err

set -x
MODEL_NAME=ecland
MODEL_OUTPUT_DIR=/scratch/nest/ecland_global_reanalysis_more_time/final_output/reanalysis/experiment/era5_v0/ecland/output/
EVALUATION_OUTPUT_DIR=/scratch/$USER/validation/
ET_MONTHLY_DIR=${EVALUATION_OUTPUT_DIR}/et_${MODEL_NAME}
ET_MONTHLY_FILE=evspsbl_${MODEL_NAME}_era5_mem_00_monthly.nc
START_YEAR=1980 # 1985 if you want to skip the first 5 years of spinup
NCPUS=8

cd "$MODEL_OUTPUT_DIR" || exit 1
ml cdo
ml python3
export PYTHONPATH=/lus/h2resw01/fws4/sb/project/C3SHydroGL/phase_3/global/exchange/model_validation_scripts/mhm_tools_env_eval/src

# ---- daily -> monthly ET (skipped if already done) ----
if [[ ! -s "$ET_MONTHLY_DIR/$ET_MONTHLY_FILE" ]]; then
    mkdir -p "$ET_MONTHLY_DIR/tmp" || exit 1
    export CDO_TIMESTAT_DATE=first   # adjust to match GLEAM/FLUXCOM time stamps
    export TMPD="$ET_MONTHLY_DIR/tmp"
    find "$MODEL_OUTPUT_DIR/mem_00" -name "evspsbl_${MODEL_NAME}_era5_mem_00_*_daily.nc" -print0 | \
        xargs -0 -P "$NCPUS" -I{} sh -c \
        'b=$(basename "$1" _daily.nc); cdo -s -O monmean "$1" "$TMPD/${b}_monthly.nc"' _ {} || exit 1

    cdo -O -z zip_4 mergetime "$TMPD"/*_monthly.nc "$ET_MONTHLY_DIR/$ET_MONTHLY_FILE" || exit 1
    rm -rf "$TMPD"
fi

# gleam
python3 -m mhm_tools._cli gridded_data_evaluation --input_path "$ET_MONTHLY_DIR" --output_dir $EVALUATION_OUTPUT_DIR/gleam --input_variable evspsbl --input_name $MODEL_NAME --input_factor 86400 --ref_path /ec/fws4/sb/project/C3SHydroGL/phase_3/global/validation_data/et/gleam_ufz/monthly_mm_day --ref_name gleam --ref_factor 1 --ref_variable E --ncpus 4 --input_file_name "$ET_MONTHLY_FILE" --ref_file_name '*.*' --global_climate

# fluxcom
python3 -m mhm_tools._cli gridded_data_evaluation --input_path "$ET_MONTHLY_DIR" --output_dir $EVALUATION_OUTPUT_DIR/fluxcom --input_variable evspsbl --input_name $MODEL_NAME --input_factor 86400 --ref_path /ec/fws4/sb/project/C3SHydroGL/phase_3/global/validation_data/et/fluxcom_ufz --ref_name fluxnet --ref_factor 0.4 --ref_variable LE --ncpus 4 --input_file_name "$ET_MONTHLY_FILE" --ref_file_name '*.*' --global_climate

# twsa
# python3 -m mhm_tools._cli twsa_evaluation --input_path ./ --output_dir $EVALUATION_OUTPUT_DIR/grace --input_variable tws --input_name $MODEL_NAME --input_format tws --ref_path /ec/fws4/sb/project/C3SHydroGL/phase_3/global/validation_data/twsa/grace --ref_name grace --ref_variable lwe_thickness --ref_format twsa --input_file_name "tws_${MODEL_NAME}_era5_mem_00_*_daily.nc" --ref_file_name GRCTellus.JPL.200204_202508.GLO.RL06.3M.MSCNv04CRI.nc --baseline_start_year 2004 --baseline_end_year 2009 --min_valid_months 24 --regions all --max_region_cell_lines 2000 --kge_vmin -0.5 --max_memory_gib 8.0

# snow
python3 -m mhm_tools._cli snow_evaluation --input_path ./ --output_dir $EVALUATION_OUTPUT_DIR/snowcci --input_variable swe --input_name $MODEL_NAME --ref_path /ec/fws4/sb/project/C3SHydroGL/phase_3/global/validation_data/snow/snowcci-merged/swe/v3.1/ --ref_name snowcci --ref_variable swe --ncpus 4 --input_file_name "swe_${MODEL_NAME}_era5_mem_00_*_daily.nc" --ref_file_name '*-ESACCI-L3C_SNOW-SWE-SSMI_DMSP-fv3.1.nc' --snow_threshold 0.0 --year_mode snow_year --snow_year_start_month 9 --regions all --gif_fps 4 --max_gif_frames 100 --max_memory_gib 4.0

echo "Done!"
